### 3. Column Renaming

```javascript
const handleColumnRename = (columnName) => {
  setActionModal({
    open: true,
    type: "rename",
    column: columnName,
    data: { newName: "" },
  });
};

const handleSaveAction = () => {
  const { type, column, data } = actionModal;

  switch (type) {
    case "rename":
      // Prevent renaming to control column names
      const reservedControlColumns = [
        "_ingest_timestamp",
        "_source_system",
        "_record_status",
        "_update_timestamp",
        "_batch_id",
        "_created_by",
        "_updated_by",
      ];

      if (reservedControlColumns.includes(data.newName)) {
        alert(
          "Cannot rename to a control column name. Control columns are reserved for system use."
        );
        return;
      }

      // Update curated attributes
      const newCuratedAttributes = curatedAttributes.map((attr) =>
        attr === column ? data.newName : attr
      );
      setCuratedAttributes(newCuratedAttributes);

      // Update mappings
      setMappings((prev) =>
        prev.map((mapping) =>
          mapping.curated === column
            ? { ...mapping, curated: data.newName }
            : mapping
        )
      );
      break;
  }

  setActionModal({ open: false, type: "", column: "", data: null });
};
```

## Performance Optimization Patterns

### 1. Debounced Auto-save

```javascript
import { useCallback } from "react";
import { debounce } from "lodash";

const EditEntityMappingsPage = ({ entityName, curatedEntities, fileData }) => {
  const debouncedAutoSave = useCallback(
    debounce(async (mappings, descriptions) => {
      await autoSaveMappings(mappings, descriptions);
    }, 1000),
    []
  );

  const updateMapping = async (idx, field, value) => {
    // Update state
    const updatedMappings = mappings.map((row, i) =>
      i === idx ? { ...row, [field]: value, autogenerated: false } : row
    );
    setMappings(updatedMappings);

    // Debounced auto-save
    if (
      field === "curated" &&
      value.trim() &&
      !updatedMappings[idx].isNewColumn
    ) {
      debouncedAutoSave(updatedMappings, columnDescriptions);
    }
  };
};
```

## Common Pitfalls and Solutions

### 1. Entity Name Encoding

**Problem**: Entity names with special characters cause API errors
**Solution**: Always URL-encode entity names

```javascript
// ❌ Wrong
const response = await axios.get(`/api/raw-schemas/${entityName}`);

// ✅ Correct
const response = await axios.get(
  `/api/raw-schemas/${encodeURIComponent(entityName)}`
);
```

### 2. State Update Race Conditions

**Problem**: Multiple rapid state updates cause inconsistent state
**Solution**: Use functional updates and proper dependency arrays

```javascript
// ❌ Wrong
setMappings(
  mappings.map((row, i) => (i === idx ? { ...row, [field]: value } : row))
);

// ✅ Correct
setMappings((prev) =>
  prev.map((row, i) => (i === idx ? { ...row, [field]: value } : row))
);
```

### 3. Memory Leaks

**Problem**: Component unmounts while API calls are in progress
**Solution**: Use cleanup functions and abort controllers

```javascript
useEffect(() => {
  const abortController = new AbortController();

  const fetchData = async () => {
    try {
      const response = await axios.get("/api/data", {
        signal: abortController.signal,
      });
      // Process response
    } catch (error) {
      if (error.name !== "CanceledError") {
        // Handle error
      }
    }
  };

  fetchData();

  return () => {
    abortController.abort();
  };
}, []);
```
